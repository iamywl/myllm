import heapq

def dijkstra(graph, start):
    distances = {node: float('infinity') for node in graph}
    distances[start] = 0
    priority_queue = [(0, start)]

    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)

        if current_distance > distances[current_node]:
            continue

        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight

            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(priority_queue, (distance, neighbor))

    return distances

        self.key = key
        self.value = value
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}
        self.head = Node(0, 0)
        self.tail = Node(0, 0)
        self.head.next = self.tail
        self.tail.prev = self.head

    def _remove(self, node):
        prev = node.prev
        nxt = node.next
        prev.next = nxt
        nxt.prev = prev

    def _add(self, node):
        prev = self.tail.prev
        prev.next = node
        self.tail.prev = node
        node.prev = prev
        node.next = self.tail

    def get(self, key: int) -> int:
        if key in self.cache:
            node = self.cache[key]
            self._remove(node)
            self._add(node)
            return node.value
        return -1

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self._remove(self.cache[key])
        node = Node(key, value)
        self._add(node)
        self.cache[key] = node

	todos    = make(map[int]Todo)
	nextID   = 1
	todoLock sync.Mutex
)

func main() {
	http.HandleFunc("/todos", getTodos)
	http.HandleFunc("/todos/", handleTodo)

	log.Println("Server started on :8080")
	log.Fatal(http.ListenAndServe(":8080", nil))

func getTodos(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	todoLock.Lock()
	defer todoLock.Unlock()

	todoSlice := make([]Todo, 0, len(todos))
	for _, todo := range todos {
		todoSlice = append(todoSlice, todo)
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(todoSlice)
}

func handleTodo(w http.ResponseWriter, r *http.Request) {
	id, err := strconv.Atoi(r.URL.Path[len("/todos/"):])
	if err != nil {
		http.Error(w, "Invalid todo ID", http.StatusBadRequest)
		return
	}

	todoLock.Lock()
	defer todoLock.Unlock()

	switch r.Method {
	case http.MethodPost:
		var todo Todo
		if err := json.NewDecoder(r.Body).Decode(&todo); err != nil {
			http.Error(w, "Invalid request payload", http.StatusBadRequest)
			return
		}
		todo.ID = nextID
		nextID++
		todos[todo.ID] = todo
		w.WriteHeader(http.StatusCreated)
		json.NewEncoder(w).Encode(todo)
	case http.MethodDelete:
		if _, exists := todos[id]; !exists {
			http.Error(w, "Todo not found", http.StatusNotFound)
			return
		}
		delete(todos, id)
		w.WriteHeader(http.StatusNoContent)
	default:
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
	}
}

class PaymentContext:

	Height float64

func (r *Rectangle) Perimeter() float64 {

type Triangle struct {
	SideA float64
	SideB float64
	SideC float64
}

func (t *Triangle) Area() float64 {
	// Heron's formula
	s := (t.SideA + t.SideB + t.SideC) / 2
	return math.Sqrt(s * (s - t.SideA) * (s - t.SideB) * (s - t.SideC))
}

func (t *Triangle) Perimeter() float64 {
	return t.SideA + t.SideB + t.SideC
}

### 메모리 누수 코드 분석 (qwen2.5-coder:32b)

- 점수: 1/2
- 키워드 체크: ticker.Stop✓, goroutine✗
- 응답 시간: 42.9s, 토큰: 635, 속도: 15.4 t/s

```go
이 코드에서 메모리 누수가 발생하는 원인은 `process` 함수가 호출될 때마다 새로운 채널이 생성되고, 이 채널로 보내진 데이터를 처리하지 못하는 고루틴이 계속 생성되기 때문입니다.

구체적으로 설명하면, `main` 함수에서 `for` 루프가 실행될 때마다 새로운 채널 `ch`가 생성되고, `process` 함수를 고루틴으로 실행하면서 이 채널을 인자로 전달합니다. 그러나 각 고루틴이 생성된 후, `ch` 채널로 "hello" 메시지가 보내지고 이후로는 채널에 다른 데이터가 들어오지 않기 때문에, `process` 함수의 `select` 문에서 채널 `ch`에서 데이터를 읽는 부분은 한 번 실행된 후 무한 루프를 돌게 됩니다. 이로 인해 각 고루틴이 계속 실행되면서 메모리가 점점 차게 됩니다.

이 문제를 해결하기 위해, 각 고루틴이 작업을 완료한 후 종료되도록 코드를 수정할 수 있습니다. 이를 위해 `process` 함수에서 메시지를 받은 후 채널을 닫는 방식을 사용할 수 있습니다. 또는 채널을 닫는 대신, 고루틴이 작업을 완료하면 `return`하여 함수를 종료할 수 있습니다.

아래는 고루틴이 작업을 완료한 후 종료되도록 수정한 코드입니다:

```go
package main

import (
    "fmt"
    "time"
)

func process(ch chan string) {
    ticker := time.NewTicker(time.Second)
    defer ticker.Stop() // ticker를 사용한 후에는 반드시 Stop() 해야 합니다.
    for {
        select {
        case msg, ok := <-ch:
            if !ok {
                return // 채널이 닫히면 고루틴 종료
            }
            fmt.Println(msg)
        case <-ticker.C:
            fmt.Println("tick")
        }
    }
}

func main() {
    for i := 0; i < 100; i++ {
        ch := make(chan string)
        go process(ch)
        ch <- "hello"
        close(ch) // 메시지를 보낸 후 채널을 닫아 고루틴이 종료되도록 합니다.
    }
    time.Sleep(time.Minute)
}
```

이제 `main` 함수에서 각 채널로 메시지를 보내고 채널을 닫은 후, `process` 함수는 메시지를 받은 후 채널이 닫혔음을 감지하여 고루틴을 종료합니다. 이를 통해 메모리 누수가 방지됩니다. 또한, `ticker`를 사용한 후에는 반드시 `Stop()` 함수를 호출하여 리소스를 해제하는 것이 좋습니다.
```

### 레이스 컨디션 수정 (qwen2.5-coder:32b)

- 점수: 3/3
- 키워드 체크: Lock✓, lock✓, threading✓

```python
이 코드에서는 `counter` 변수에 대한 접근이 여러 스레드에서 동시에 일어나는 레이스 컨디션이 발생합니다. 각 스레드가 `counter` 변수를 읽고, 증가시키고, 다시 저장하는 과정에서 다른 스레드가 동일한 작업을 수행할 수 있습니다. 이로 인해 `counter`의 최종 값이 예상한 1,000,000이 되지 않을 수 있습니다.

이 문제를 해결하기 위해서는 `counter` 변수에 대한 접근을 동기화해야 합니다. 이를 위해 `threading.Lock` 객체를 사용할 수 있습니다.

아래는 수정된 코드입니다:

```python
import threading

counter = 0
lock = threading.Lock()  # 락 객체 생성

def increment():
    global counter
    for _ in range(100000):
        with lock:  # 락을 획득하고 작업을 수행한 후 자동으로 해제
            counter += 1

threads = [threading.Thread(target=increment) for _ in range(10)]
for t in threads:
    t.start()
for t in threads:
    t.join()
print(counter)  # 이제 1000000이 출력됨
```

수정된 코드에서 `lock` 객체를 사용하여 `counter` 변수에 대한 접근을 동기화했습니다. `with lock:` 구문은 락을 획득한 스레드만이 `counter` 변수를 안전하게 변경할 수 있도록 보장합니다. 다른 스레드들은 락이 해제될 때까지 대기하게 됩니다. 이렇게 하면 모든 스레드가 `counter` 변수를 정확하게 100,000번씩 증가시키므로, 최종 값은 1,000,000이 됩니다.
